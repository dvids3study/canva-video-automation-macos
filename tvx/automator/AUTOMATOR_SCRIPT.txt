export PATH="/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin:/usr/sbin:/sbin"
BASE="$HOME/Library/Application Support/VIDEOTV"

es_nombre_valido() {
    local nombre="$1" lower
    lower="$(printf '%s' "$nombre" | tr '[:upper:]' '[:lower:]')"

    case "$lower" in
        *.download|*.crdownload|*.part|*.tmp) return 1 ;;
    esac

    # El resultado final de TVX no se reprocesa.
    case "$lower" in
        videotv.mp4|video_tv.mp4) return 1 ;;
    esac

    [[ "$lower" =~ ^(videotv|video_tv)$ ]] && return 0
    [[ "$lower" =~ ^(videotv|video_tv)[[:space:]]\([0-9]+\)$ ]] && return 0
    [[ "$lower" =~ ^(videotv|video_tv)[[:space:]][0-9]+$ ]] && return 0
    [[ "$lower" =~ ^(videotv|video_tv)\.zip$ ]] && return 0
    [[ "$lower" =~ ^(videotv|video_tv)[[:space:]]\([0-9]+\)\.zip$ ]] && return 0
    [[ "$lower" =~ ^(videotv|video_tv)[[:space:]][0-9]+\.zip$ ]] && return 0
    [[ "$lower" =~ ^(videotv|video_tv)\.zip[[:space:]]?\([0-9]+\)$ ]] && return 0

    # Nombres como "Chamba — VIDEOTV.mp4"
    case "$lower" in
        *videotv*|*video_tv*) return 0 ;;
    esac
    return 1
}

for item in "$@"; do
    nombre="$(basename "$item")"

    # Nuestro resultado final y paquetes TVX nunca coinciden con estas reglas.
    [ "$nombre" = "VIDEOTV.mp4" ] && continue

    if es_nombre_valido "$nombre"; then
        if [ ! -x "$BASE/VIDEOTV_Watcher.sh" ]; then
            echo "TVX no está instalado: falta $BASE/VIDEOTV_Watcher.sh" >&2
            exit 1
        fi
        nohup "$BASE/VIDEOTV_Watcher.sh" "$item" >/dev/null 2>&1 &
        exit 0
    fi
done

exit 0
